   235→   235→const CountdownTimer = memo(function CountdownTimer({ expiresAt, compact = false }: { expiresAt: string; compact?: boolean }) {
   236→   236→  const [timeLeft, setTimeLeft] = useState('')
   237→   237→
   238→   238→  useEffect(() => {
   239→   239→    const update = () => {
   240→   240→      const diff = new Date(expiresAt).getTime() - Date.now()
   241→   241→      if (diff <= 0) {
   242→   242→        setTimeLeft('Expired')
   243→   243→        return
   244→   244→      }
   245→   245→      const hours = Math.floor(diff / 3600000)
   246→   246→      const mins = Math.floor((diff % 3600000) / 60000)
   247→   247→      const secs = Math.floor((diff % 60000) / 1000)
   248→   248→
   249→   249→      if (hours > 0) {
   250→   250→        setTimeLeft(`${hours}h ${mins}m`)
   251→   251→      } else if (mins > 0) {
   252→   252→        setTimeLeft(`${mins}m ${secs}s`)
   253→   253→      } else {
   254→   254→        setTimeLeft(`${secs}s`)
   255→   255→      }
   256→   256→    }
   257→   257→    update()
   258→   258→    const interval = setInterval(update, 1000)
   259→   259→    return () => clearInterval(interval)
   260→   260→  }, [expiresAt])
   261→   261→
   262→   262→  const isUrgent = timeLeft !== 'Expired' && !timeLeft.includes('h')
   263→   263→
   264→   264→  return (
   265→   265→    <span className={`inline-flex items-center gap-1 ${compact ? 'text-xs' : 'text-sm'} font-bold ${
   266→   266→      timeLeft === 'Expired' ? 'text-[#70787d]' : isUrgent ? 'text-[#FB923C] animate-pulse-urgent' : 'text-[#0d6683]'
   267→   267→    }`}>
   268→   268→      <Clock className={compact ? 'w-3 h-3' : 'w-3.5 h-3.5'} />
   269→   269→      {timeLeft}
   270→   270→    </span>
   271→   271→  )
   272→   272→})

   629→   629→function DealDetailView() {
   630→   630→  const { viewParams, goBack, setShowAuthModal, navigate } = useAppStore()
   631→   631→  const [deal, setDeal] = useState<Deal | null>(null)
   632→   632→  const [loading, setLoading] = useState(true)
   633→   633→  const [claiming, setClaiming] = useState(false)
   634→   634→  const [claimed, setClaimed] = useState(false)
   635→   635→  const [order, setOrder] = useState<Order | null>(null)
   636→   636→  const { user, isAuthenticated } = useAuthStore()
   637→   637→
   638→   638→  useEffect(() => {
   639→   639→    if (!viewParams.id) return
   640→   640→    setLoading(true)
   641→   641→    apiFetch<Deal>(`/api/deals/${viewParams.id}`).then((res) => {
   642→   642→      if (res.success && res.data) setDeal(res.data)
   643→   643→    }).finally(() => setLoading(false))
   644→   644→  }, [viewParams.id])
   645→   645→
   646→   646→  const handleClaim = async () => {
   647→   647→    if (!deal) return
   648→   648→    if (!isAuthenticated || !user) {
   649→   649→      setShowAuthModal(true)
   650→   650→      return
   651→   651→    }
   652→   652→    setClaiming(true)
   653→   653→    try {
   654→   654→      // Step 1: Claim (create reservation)
   655→   655→      const claimRes = await apiFetch<{ reservation: { id: string } }>(`/api/deals/${deal.id}/claim`, {
   656→   656→        method: 'POST',
   657→   657→        body: JSON.stringify({ quantity: 1 }),
   658→   658→      })
   659→   659→      if (!claimRes.success) {
   660→   660→        toast.error(claimRes.error || 'Failed to claim deal')
   661→   661→        return
   662→   662→      }
   663→   663→
   664→   664→      const reservationId = claimRes.data?.reservation?.id
   665→   665→
   666→   666→      // Step 2: Confirm reservation → Order
   667→   667→      const confirmRes = await apiFetch<Order>(`/api/deals/${deal.id}/confirm`, {
   668→   668→        method: 'POST',
   669→   669→        body: JSON.stringify({ reservationId }),
   670→   670→      })
   671→   671→      if (confirmRes.success && confirmRes.data) {
   672→   672→        setOrder(confirmRes.data)
   673→   673→        setClaimed(true)
   674→   674→        toast.success('Deal claimed! Check your orders for the QR code.')
   675→   675→
   676→   676→        // Refresh deal to show updated stock
   677→   677→        const refreshRes = await apiFetch<Deal>(`/api/deals/${deal.id}`)
   678→   678→        if (refreshRes.success && refreshRes.data) setDeal(refreshRes.data)
   679→   679→      } else {
   680→   680→        toast.error(confirmRes.error || 'Failed to confirm claim')
   681→   681→      }
   682→   682→    } finally {
   683→   683→      setClaiming(false)
   684→   684→    }
   685→   685→  }
   686→   686→
   687→   687→  if (loading) {
   688→   688→    return (
   689→   689→      <div className="pb-28">
   690→   690→        <Skeleton className="aspect-[16/10] rounded-none" />
   691→   691→        <div className="px-5 py-4 space-y-4">

   695→   695→        </div>
   696→   696→      </div>
   697→   697→    )
   698→   698→  }
   699→   699→
   700→   700→  if (!deal) {

   706→   706→  }

   709→   709→

   719→   719→        {deal.imageUrl ? (
