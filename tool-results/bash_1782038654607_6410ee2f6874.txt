    55→// ============================================
    56→// API Helper (Bearer-token auth with transparent refresh on 401)
    57→// ============================================
    58→// Auth uses Bearer tokens stored in localStorage (via the auth store) as the
    59→// PRIMARY mechanism, because cookies are unreliable in preview iframes
    60→// (third-party cookie blocking). The server's getAuthUser() checks the
    61→// Authorization header first, then falls back to cookies.
    62→//
    63→// Flow: every request gets `Authorization: Bearer <accessToken>`. On 401 we
    64→// call /api/auth/refresh (sending the refresh token in the body), store the
    65→// new tokens, and retry once. Concurrent 401s share a single refresh.
    66→let refreshPromise: Promise<boolean> | null = null
    67→
    68→async function refreshAccessToken(): Promise<boolean> {
    69→  if (refreshPromise) return refreshPromise
    70→  refreshPromise = (async () => {
    71→    try {
    72→      const { refreshToken } = useAuthStore.getState()
    73→      const res = await fetch('/api/auth/refresh', {
    74→        method: 'POST',
    75→        headers: { 'Content-Type': 'application/json' },
    76→        body: JSON.stringify({ refreshToken }),
    77→      })
    78→      if (!res.ok) return false
    79→      const json = await res.json()
    80→      if (json.success && json.tokens) {
    81→        useAuthStore.getState().setTokens(json.tokens)
    82→        return true
    83→      }
    84→      return false
    85→    } catch {
    86→      return false
    87→    } finally {
    88→      refreshPromise = null
    89→    }
    90→  })()
    91→  return refreshPromise
    92→}

    94→async function apiFetch<T>(path: string, options?: RequestInit): Promise<{ success: boolean; data?: T; error?: string; tokens?: { accessToken: string; refreshToken: string } }> {
    95→  const buildHeaders = (opts?: RequestInit) => {
    96→    const headers: Record<string, string> = {
    97→      'Content-Type': 'application/json',
    98→      ...(opts?.headers as Record<string, string> | undefined),
    99→    }
   100→    // Attach Bearer token from the auth store (localStorage-backed)
   101→    const { accessToken } = useAuthStore.getState()
   102→    if (accessToken && !headers['Authorization'] && !headers['authorization']) {
   103→      headers['Authorization'] = `Bearer ${accessToken}`
   104→    }
   105→    return headers
   106→  }
   107→
   108→  try {
   109→    let res = await fetch(path, { ...options, headers: buildHeaders(options) })
   110→
   111→    // If the access token has expired, transparently refresh + retry once.
   112→    // Skip this for the refresh/logout endpoints themselves to avoid loops.
   113→    if (
   114→      res.status === 401 &&
   115→      !path.includes('/api/auth/refresh') &&
   116→      !path.includes('/api/auth/logout')
   117→    ) {
   118→      const refreshed = await refreshAccessToken()
   119→      if (refreshed) {
   120→        // Retry the original request with the fresh Bearer token
   121→        res = await fetch(path, { ...options, headers: buildHeaders(options) })
   122→      } else {
   123→        return { success: false, error: 'Session expired. Please sign in again.' }
   124→      }
   125→    }
   126→
   127→    const text = await res.text()
   128→    try {
   129→      return JSON.parse(text)
   130→    } catch {
   131→      return { success: res.ok, error: res.ok ? undefined : 'Request failed' }
   132→    }
   133→  } catch {
   134→    return { success: false, error: 'Network error' }
   135→  }
   136→}

   356→// COUNTDOWN TIMER COMPONENT
   357→// ============================================
   358→const CountdownTimer = memo(function CountdownTimer({ expiresAt, compact = false }: { expiresAt: string; compact?: boolean }) {
   359→  const [timeLeft, setTimeLeft] = useState('')
   360→
   361→  useEffect(() => {
   362→    const update = () => {
   363→      const diff = new Date(expiresAt).getTime() - Date.now()
   364→      if (diff <= 0) {
   365→        setTimeLeft('Expired')
   366→        return
   367→      }
   368→      const hours = Math.floor(diff / 3600000)
   369→      const mins = Math.floor((diff % 3600000) / 60000)
   370→      const secs = Math.floor((diff % 60000) / 1000)
   371→
   372→      if (hours > 0) {
   373→        setTimeLeft(`${hours}h ${mins}m`)
   374→      } else if (mins > 0) {
   375→        setTimeLeft(`${mins}m ${secs}s`)
   376→      } else {
   377→        setTimeLeft(`${secs}s`)
   378→      }
   379→    }
   380→    update()
   381→    const interval = setInterval(update, 1000)
   382→    return () => clearInterval(interval)
   383→  }, [expiresAt])
   384→
   385→  const isUrgent = timeLeft !== 'Expired' && !timeLeft.includes('h')
   386→
   387→  return (
   388→    <span className={`inline-flex items-center gap-1 ${compact ? 'text-xs' : 'text-sm'} font-bold ${
   389→      timeLeft === 'Expired' ? 'text-[#717971]' : isUrgent ? 'text-[#FB923C] animate-pulse-urgent' : 'text-[#6CB4EE]'
   390→    }`}>
   391→      <Clock className={compact ? 'w-3 h-3' : 'w-3.5 h-3.5'} />
   392→      {timeLeft}
   393→    </span>
   394→  )
   395→})

   507→
   508→  useEffect(() => {
   509→    const update = () => {
   510→      const diff = new Date(expiresAt).getTime() - Date.now()
   511→      if (diff <= 0) { setTimeLeft('Expired'); return }
   512→      const hours = Math.floor(diff / 3600000)
   513→      const mins = Math.floor((diff % 3600000) / 60000)
   514→      const secs = Math.floor((diff % 60000) / 1000)
   515→      if (hours > 0) setTimeLeft(`${hours}h ${mins}m`)
   516→      else if (mins > 0) setTimeLeft(`${mins}m ${secs}s`)
   517→      else setTimeLeft(`${secs}s`)
   518→    }
   519→    update()
   520→    const interval = setInterval(update, 1000)
   521→    return () => clearInterval(interval)
   522→  }, [expiresAt])
   523→
   524→  const isUrgent = timeLeft !== 'Expired' && !timeLeft.includes('h')
   525→
   526→  return (
   527→    <span className={`inline-flex items-center gap-0.5 text-[10px] font-bold px-1.5 py-0.5 rounded-md backdrop-blur-sm ${
   528→      timeLeft === 'Expired' 
   529→        ? 'bg-black/40 text-white/70' 
   530→        : isUrgent 
   531→        ? 'bg-[#FB923C]/90 text-white animate-pulse-urgent' 
   532→        : 'bg-black/40 text-white/90'
   533→    }`}>
   534→      <Clock className="w-2.5 h-2.5" />
   535→      {timeLeft}
   536→    </span>
   537→  )
   538→})

   541→// QR CODE IMAGE COMPONENT (generates real QR from qrCode string)
   542→// ============================================
   543→function QRCodeImage({ qrCode }: { qrCode: string }) {
   544→  const [src, setSrc] = useState<string | null>(null)
   545→  const [error, setError] = useState(false)
   546→
   547→  useEffect(() => {
   548→    let cancelled = false
   549→    import('qrcode').then((QRCode) => {
   550→      if (cancelled) return
   551→      QRCode.toDataURL(qrCode, {
   552→        width: 220,
   553→        margin: 2,
   554→        color: { dark: '#1a1c1e', light: '#ffffff' },
   555→        errorCorrectionLevel: 'M',
   556→      }).then((url: string) => {
   557→        if (!cancelled) setSrc(url)
   558→      }).catch(() => {
   559→        if (!cancelled) setError(true)
   560→      })
   561→    }).catch(() => {
   562→      if (!cancelled) setError(true)
   563→    })
   564→    return () => { cancelled = true }
   565→  }, [qrCode])
   566→
   567→  if (error || !src) {
   568→    return (
   569→      <div className="w-[220px] h-[220px] flex items-center justify-center bg-[#f0f4f2] rounded-xl">
   570→        <QrCode className="w-20 h-20 text-[#6CB4EE]" />
   571→      </div>
   572→    )
   573→  }
   574→
   575→  return (
   576→    <Image
   577→      src={src}
   578→      alt="QR Code for order pickup"
   579→      width={220}
   580→      height={220}
   581→      className="rounded-lg"
   582→      unoptimized
   583→    />
   584→  )
   585→}

   587→// ============================================
   588→// IMAGE UPLOADER COMPONENT
   589→// High-tech drag-and-drop + camera upload with auto-resize
   590→// ============================================
   591→interface ImageUploaderProps {
   592→  /** Which preset group to use for auto-sizing */
   593→  group: 'profile' | 'vendor_logo' | 'vendor_banner' | 'deal' | 'notification'
   594→  /** Current image URL (for preview) */
   595→  currentUrl?: string | null
   596→  /** Called with the new URL after successful upload */
   597→  onUploadComplete: (url: string, urls: Record<string, string>) => void
   598→  /** Optional label */
   599→  label?: string
   600→  /** Optional size hint */
   601→  sizeHint?: string
   602→  /** Circle crop for avatars */
   603→  circular?: boolean
   604→  /** Compact mode (no label, smaller) */
   605→  compact?: boolean
   606→}
   607→
   608→function ImageUploader({
   609→  group,
   610→  currentUrl,
   611→  onUploadComplete,
   612→  label,
   613→  sizeHint,
   614→  circular = false,
   615→  compact = false,
   616→}: ImageUploaderProps) {
   617→  const [dragging, setDragging] = useState(false)
   618→  const [uploading, setUploading] = useState(false)
   619→  const [progress, setProgress] = useState(0)
   620→  const [preview, setPreview] = useState<string | null>(null)
   621→  const [fileError, setFileError] = useState<string | null>(null)
   622→  const fileInputRef = useRef<HTMLInputElement>(null)
   623→
   624→  const handleFile = async (file: File) => {
   625→    const { validateImageFile } = await import('@/lib/image-utils')
   626→    const validation = validateImageFile(file)
   627→    if (!validation.valid) {
   628→      setFileError(validation.error ?? 'Invalid file')
   629→      return
   630→    }
   631→    setFileError(null)
   632→
   633→    // Show preview immediately
   634→    const reader = new FileReader()
   635→    reader.onload = (e) => setPreview(e.target?.result as string)
   636→    reader.readAsDataURL(file)
   637→
   638→    setUploading(true)
   639→    setProgress(0)
   640→
   641→    try {
   642→      const { quickUpload } = await import('@/lib/image-utils')
   643→      const { uploadImageVariants } = await import('@/lib/image-utils')
   644→      const result = await uploadImageVariants(file, group, {
   645→        onProgress: setProgress,
   646→      })
   647→
   648→      // Get the primary URL for this group
   649→      const primaryKey: Record<string, string> = {
   650→        profile: 'avatar',
   651→        vendor_logo: 'logo',
   652→        vendor_banner: 'banner',
   653→        deal: 'medium',
   654→        notification: 'notif',
   655→      }
   656→      const primaryUrl = result.urls[primaryKey[group]] || result.originalUrl
   657→      onUploadComplete(primaryUrl, result.urls)
   658→      toast.success('Image uploaded!')
   659→    } catch (err) {
   660→      toast.error(err instanceof Error ? err.message : 'Upload failed')
   661→      setPreview(null)
   662→    } finally {
   663→      setUploading(false)
   664→      setProgress(0)
   665→    }
   666→  }
   667→
   668→  const handleDrop = (e: React.DragEvent) => {
   669→    e.preventDefault()
   670→    setDragging(false)
   671→    const file = e.dataTransfer.files?.[0]
   672→    if (file) handleFile(file)
   673→  }
   674→
   675→  const handleDragOver = (e: React.DragEvent) => {
   676→    e.preventDefault()
   677→    setDragging(true)
   678→  }
   679→
   680→  const handleDragLeave = () => setDragging(false)
   681→
   682→  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   683→    const file = e.target.files?.[0]
   684→    if (file) handleFile(file)
   685→  }
   686→
   687→  const displayUrl = preview || currentUrl
   688→
   689→  if (compact) {
   690→    return (
   691→      <div>
   692→        {fileError && (
   693→          <p className="text-xs text-[#EF4444] font-medium mb-1.5 flex items-center gap-1">
   694→            <AlertCircle className="w-3 h-3 flex-shrink-0" />
   695→            {fileError}
   696→          </p>
   697→        )}
   698→        <div
   699→          onDrop={handleDrop}
   700→          onDragOver={handleDragOver}
   701→          onDragLeave={handleDragLeave}
   702→          onClick={() => !uploading && fileInputRef.current?.click()}
   703→          className={`relative cursor-pointer group ${circular ? 'rounded-full' : 'rounded-xl'} overflow-hidden ${dragging ? 'ring-2 ring-[#6CB4EE]' : ''}`}
   704→        >
   705→          {displayUrl ? (
   706→          <div className={`relative ${circular ? 'w-20 h-20' : 'w-24 h-16'}`}>
   707→            <Image src={displayUrl} alt="Upload" fill className={`${circular ? 'rounded-full' : 'rounded-xl'} object-cover`} unoptimized />
   708→            <div className="absolute inset-0 bg-black/0 group-hover:bg-black/30 transition-colors flex items-center justify-center">
   709→              {uploading ? (
   710→                <div className="w-8 h-8 rounded-full border-2 border-white border-t-transparent animate-spin" />
   711→              ) : (
   712→                <Camera className="w-5 h-5 text-white opacity-0 group-hover:opacity-100 transition-opacity" />
   713→              )}
   714→            </div>
   715→          </div>
   716→        ) : (
   717→          <div className={`${circular ? 'w-20 h-20' : 'w-24 h-16'} flex items-center justify-center bg-[#f0f4f2] ${circular ? 'rounded-full' : 'rounded-xl'} border-2 border-dashed border-[#c1c9c0]`}>
   718→            {uploading ? (
   719→              <div className="w-6 h-6 rounded-full border-2 border-[#6CB4EE] border-t-transparent animate-spin" />
   720→            ) : (
   721→              <PlusCircle className="w-5 h-5 text-[#717971]" />
   722→            )}
   723→          </div>
   724→        )}
   725→          <input ref={fileInputRef} type="file" accept="image/*" onChange={handleInputChange} className="hidden" />
   726→        </div>
   727→      </div>
   728→    )
   729→  }
   730→
   731→  return (
   732→    <div className="space-y-1.5">
   733→      {label && <Label className="text-[11px] font-bold text-[#414841]">{label}</Label>}
   734→      {fileError && (
   735→        <p className="text-xs text-[#EF4444] font-medium mb-1.5 flex items-center gap-1">
   736→          <AlertCircle className="w-3 h-3 flex-shrink-0" />
   737→          {fileError}
   738→        </p>
   739→      )}
   740→      <div
   741→        onDrop={handleDrop}
   742→        onDragOver={handleDragOver}
   743→        onDragLeave={handleDragLeave}
   744→        onClick={() => !uploading && fileInputRef.current?.click()}
   745→        className={`relative cursor-pointer group border-2 border-dashed rounded-xl transition-all ${
   746→          dragging ? 'border-[#6CB4EE] bg-[#6CB4EE]/5' : 'border-[#d7ddd9] bg-[#f8faf9] hover:border-[#8FC5E8]'
   747→        } ${displayUrl ? 'p-0 overflow-hidden' : 'p-6'}`}
   748→      >
   749→        {displayUrl ? (
   750→          <div className="relative">
   751→            <div className={`${circular ? 'w-24 h-24 mx-auto rounded-full' : 'w-full h-40'} overflow-hidden`}>
   752→              <Image src={displayUrl} alt="Preview" fill className={`${circular ? 'rounded-full' : ''} object-cover`} unoptimized />
   753→            </div>
   754→            <div className="absolute inset-0 bg-black/0 group-hover:bg-black/30 transition-colors flex items-center justify-center gap-2">
   755→              {uploading ? (
   756→                <div className="flex flex-col items-center">
   757→                  <div className="w-8 h-8 rounded-full border-2 border-white border-t-transparent animate-spin mb-2" />
   758→                  <span className="text-white text-xs font-bold">{progress}%</span>
   759→                </div>
   760→              ) : (
   761→                <div className="flex items-center gap-2 opacity-0 group-hover:opacity-100 transition-opacity">
   762→                  <Camera className="w-5 h-5 text-white" />
   763→                  <span className="text-white text-sm font-bold">Change Photo</span>
   764→                </div>
   765→              )}
   766→            </div>
   767→          </div>
   768→        ) : (
   769→          <div className="text-center">
   770→            {uploading ? (
   771→              <div className="flex flex-col items-center">
   772→                <div className="w-10 h-10 rounded-full border-3 border-[#6CB4EE] border-t-transparent animate-spin mb-2" />
   773→                <p className="text-xs text-[#717971]">Uploading... {progress}%</p>
   774→                <Progress value={progress} className="w-32 h-1.5 mt-2" />
   775→              </div>
   776→            ) : (
   777→              <>
   778→                <div className="w-12 h-12 rounded-full bg-[#6CB4EE]/10 flex items-center justify-center mx-auto mb-2">
   779→                  <Camera className="w-6 h-6 text-[#6CB4EE]" />
   780→                </div>
   781→                <p className="text-sm font-bold text-[#1a1c1e]">
   782→                  {dragging ? 'Drop image here' : 'Tap to upload'}
   783→                </p>
   784→                <p className="text-[11px] text-[#717971] mt-0.5">
   785→                  Drag & drop or tap to browse
   786→                </p>
   787→                {sizeHint && (
   788→                  <p className="text-[10px] text-[#717971] mt-1">{sizeHint}</p>
   789→                )}
   790→              </>
   791→            )}
   792→          </div>
   793→        )}
   794→        <input ref={fileInputRef} type="file" accept="image/*" onChange={handleInputChange} className="hidden" />
   795→      </div>
   796→      {uploading && !displayUrl && (
   797→        <Progress value={progress} className="h-1.5" />
   798→      )}
   799→    </div>
   800→  )
