   233→   233→// ============================================
   234→   234→const CountdownTimer = memo(function CountdownTimer({ expiresAt, compact = false }: { expiresAt: string; compact?: boolean }) {
   235→   235→  const [timeLeft, setTimeLeft] = useState('')
   236→   236→
   237→   237→  useEffect(() => {
   238→   238→    const update = () => {
   239→   239→      const diff = new Date(expiresAt).getTime() - Date.now()
   240→   240→      if (diff <= 0) {
   241→   241→        setTimeLeft('Expired')
   242→   242→        return
   243→   243→      }
   244→   244→      const hours = Math.floor(diff / 3600000)
   245→   245→      const mins = Math.floor((diff % 3600000) / 60000)
   246→   246→      const secs = Math.floor((diff % 60000) / 1000)
   247→   247→
   248→   248→      if (hours > 0) {
   249→   249→        setTimeLeft(`${hours}h ${mins}m`)
   250→   250→      } else if (mins > 0) {
   251→   251→        setTimeLeft(`${mins}m ${secs}s`)
   252→   252→      } else {
   253→   253→        setTimeLeft(`${secs}s`)
   254→   254→      }
   255→   255→    }
   256→   256→    update()
   257→   257→    const interval = setInterval(update, 1000)
   258→   258→    return () => clearInterval(interval)
   259→   259→  }, [expiresAt])
   260→   260→
   261→   261→  const isUrgent = timeLeft !== 'Expired' && !timeLeft.includes('h')
   262→   262→
   263→   263→  return (
   264→   264→    <span className={`inline-flex items-center gap-1 ${compact ? 'text-xs' : 'text-sm'} font-bold ${
   265→   265→      timeLeft === 'Expired' ? 'text-[#717971]' : isUrgent ? 'text-[#FB923C] animate-pulse-urgent' : 'text-[#00B14F]'
   266→   266→    }`}>
   267→   267→      <Clock className={compact ? 'w-3 h-3' : 'w-3.5 h-3.5'} />
   268→   268→      {timeLeft}
   269→   269→    </span>
   270→   270→  )
   271→   271→})
