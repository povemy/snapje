    67→// ============================================
    68→// API Helper (Bearer-token auth with transparent refresh on 401)
    69→// ============================================
    70→// Auth uses Bearer tokens stored in localStorage (via the auth store) as the
    71→// PRIMARY mechanism, because cookies are unreliable in preview iframes
    72→// (third-party cookie blocking). The server's getAuthUser() checks the
    73→// Authorization header first, then falls back to cookies.
    74→//
    75→// CRITICAL FIX (tokens in localStorage): only the short-lived accessToken
    76→// (15 min) is kept in localStorage. The refresh token lives ONLY in the
    77→// httpOnly cookie, so refresh requests use `credentials: 'include'` and let
    78→// the browser attach the cookie automatically. This prevents refresh-token
    79→// theft via XSS.
    80→//
    81→// Flow: every request gets `Authorization: Bearer <accessToken>`. On 401 we
    82→// call /api/auth/refresh (sending credentials so the cookie is attached),
    83→// store the new accessToken, and retry once. Concurrent 401s share a single refresh.
    84→let refreshPromise: Promise<boolean> | null = null
    85→
    86→async function refreshAccessToken(): Promise<boolean> {
    87→  if (refreshPromise) return refreshPromise
    88→  refreshPromise = (async () => {
    89→    try {
    90→      // CRITICAL FIX: do NOT send a refresh token in the body — it's in the
    91→      // httpOnly cookie. `credentials: 'include'` ensures the cookie is sent.
    92→      const res = await fetch('/api/auth/refresh', {
    93→        method: 'POST',
    94→        headers: { 'Content-Type': 'application/json' },
    95→        credentials: 'include',
    96→      })
    97→      if (!res.ok) return false
    98→      const json = await res.json()
    99→      if (json.success && json.tokens && json.tokens.accessToken) {
   100→        useAuthStore.getState().setTokens({ accessToken: json.tokens.accessToken })
   101→        return true
   102→      }
   103→      return false
   104→    } catch {
   105→      return false
   106→    } finally {
   107→      refreshPromise = null
   108→    }
   109→  })()
   110→  return refreshPromise
   111→}

   113→async function apiFetch<T>(path: string, options?: RequestInit): Promise<{ success: boolean; data?: T; error?: string; tokens?: { accessToken?: string } }> {
   114→  const buildHeaders = (opts?: RequestInit) => {
   115→    const headers: Record<string, string> = {
   116→      'Content-Type': 'application/json',
   117→      ...(opts?.headers as Record<string, string> | undefined),
   118→    }
   119→    // Attach Bearer token from the auth store (localStorage-backed)
   120→    const { accessToken } = useAuthStore.getState()
   121→    if (accessToken && !headers['Authorization'] && !headers['authorization']) {
   122→      headers['Authorization'] = `Bearer ${accessToken}`
   123→    }
   124→    return headers
   125→  }
   126→
   127→  try {
   128→    // CRITICAL FIX: always include credentials so the httpOnly refresh_token
   129→    // cookie is available to the server on 401 → /api/auth/refresh flows.
   130→    let res = await fetch(path, { ...options, headers: buildHeaders(options), credentials: 'include' })
   131→
   132→    // If the access token has expired, transparently refresh + retry once.
   133→    // Skip this for the refresh/logout endpoints themselves to avoid loops.
   134→    if (
   135→      res.status === 401 &&
   136→      !path.includes('/api/auth/refresh') &&
   137→      !path.includes('/api/auth/logout')
   138→    ) {
   139→      const refreshed = await refreshAccessToken()
   140→      if (refreshed) {
   141→        // Retry the original request with the fresh Bearer token
   142→        res = await fetch(path, { ...options, headers: buildHeaders(options), credentials: 'include' })
   143→      } else {
   144→        return { success: false, error: 'Session expired. Please sign in again.' }
   145→      }
   146→    }
   147→
   148→    const text = await res.text()
   149→    try {
   150→      return JSON.parse(text)
   151→    } catch {
   152→      return { success: res.ok, error: res.ok ? undefined : 'Request failed' }
   153→    }
   154→  } catch {
   155→    return { success: false, error: 'Network error' }
   156→  }
   157→}

   379→const CountdownTimer = memo(function CountdownTimer({ expiresAt, compact = false }: { expiresAt: string; compact?: boolean }) {
   380→  const [timeLeft, setTimeLeft] = useState('')
   381→
   382→  useEffect(() => {
   383→    const update = () => {
   384→      const diff = new Date(expiresAt).getTime() - Date.now()
   385→      if (diff <= 0) {
   386→        setTimeLeft('Expired')
   387→        return
   388→      }
   389→      const hours = Math.floor(diff / 3600000)
   390→      const mins = Math.floor((diff % 3600000) / 60000)
   391→      const secs = Math.floor((diff % 60000) / 1000)
   392→
   393→      if (hours > 0) {
   394→        setTimeLeft(`${hours}h ${mins}m`)
   395→      } else if (mins > 0) {
   396→        setTimeLeft(`${mins}m ${secs}s`)
   397→      } else {
   398→        setTimeLeft(`${secs}s`)
   399→      }
   400→    }
   401→    update()
   402→    const interval = setInterval(update, 1000)
   403→    return () => clearInterval(interval)
   404→  }, [expiresAt])
   405→
   406→  const isUrgent = timeLeft !== 'Expired' && !timeLeft.includes('h')
   407→
   408→  return (
   409→    <span className={`inline-flex items-center gap-1 ${compact ? 'text-xs' : 'text-sm'} font-bold ${
   410→      timeLeft === 'Expired' ? 'text-[#717971]' : isUrgent ? 'text-[#E53935] animate-pulse-urgent' : 'text-[#E53935]'
   411→    }`}>
   412→      <Clock className={compact ? 'w-3 h-3' : 'w-3.5 h-3.5'} />
   413→      {timeLeft}
   414→    </span>
   415→  )
   416→})

   524→// COUNTDOWN TIMER OVERLAY (compact, for cards)

   526→const CountdownTimerOverlay = memo(function CountdownTimerOverlay({ expiresAt }: { expiresAt: string }) {
   527→  const [timeLeft, setTimeLeft] = useState('')
   528→
   529→  useEffect(() => {
   530→    let interval: ReturnType<typeof setInterval> | null = null
   531→    const update = () => {
   532→      const diff = new Date(expiresAt).getTime() - Date.now()
   533→      if (diff <= 0) {
   534→        setTimeLeft('Expired')
   535→        // LOW 9: once the deadline has passed, stop firing setInterval — the
   536→        // value will never change again, so continuing to tick every second is
   537→        // pure CPU/battery waste (especially on mobile).
   538→        if (interval) {
   539→          clearInterval(interval)
   540→          interval = null
   541→        }
   542→        return
   543→      }
   544→      const hours = Math.floor(diff / 3600000)
   545→      const mins = Math.floor((diff % 3600000) / 60000)
   546→      const secs = Math.floor((diff % 60000) / 1000)
   547→      if (hours > 0) setTimeLeft(`${hours}h ${mins}m`)
   548→      else if (mins > 0) setTimeLeft(`${mins}m ${secs}s`)
   549→      else setTimeLeft(`${secs}s`)
   550→    }
   551→    update()
   552→    interval = setInterval(update, 1000)
   553→    return () => {
   554→      if (interval) clearInterval(interval)
   555→    }
   556→  }, [expiresAt])
   557→
   558→  const isUrgent = timeLeft !== 'Expired' && !timeLeft.includes('h')
   559→
   560→  return (
   561→    <span className={`inline-flex items-center gap-0.5 text-[10px] font-bold px-1.5 py-0.5 rounded-md backdrop-blur-sm ${
   562→      timeLeft === 'Expired' 
   563→        ? 'bg-black/40 text-white/70' 
   564→        : isUrgent 
   565→        ? 'bg-[#E53935]/90 text-white animate-pulse-urgent' 
   566→        : 'bg-black/40 text-white/90'
   567→    }`}>
   568→      <Clock className="w-2.5 h-2.5" />
   569→      {timeLeft}
   570→    </span>
   571→  )
   572→})

   882→      const raw = window.localStorage.getItem('snapje_settings')
   883→      if (!raw) return 5
   884→      const parsed = JSON.parse(raw)
   885→      const n = Number(parsed?.dealAlertRadius)

   887→    } catch {
   888→      return 5
   889→    }

   892→  const fetchDeals = useCallback(async (signal?: AbortSignal, opts?: { withRadius?: boolean }) => {
   893→    setLoading(true)

   921→        setDeals([])
   922→        setLoading(false)

   932→      // ISSUE 2: if "Near me" is on but the filter returned zero deals, fall

   941→        if (selectedCategory && selectedCategory !== 'All') fallbackParams.set('category', selectedCategory)
   942→        if (debouncedSearch) fallbackParams.set('search', debouncedSearch)
   943→        try {
   944→          const fb = await apiFetch<{ deals: Deal[]; total: number }>(`/api/deals?${fallbackParams}`)
   945→          if (!signal?.aborted && fb.success && fb.data) {
   946→            setDeals(fb.data.deals || [])
   947→          }
   948→        } catch (err2) {
   949→          console.error('[FoodieHomeView] fallback fetchDeals failed:', err2)
   950→        }
   951→      } else {
   952→        setDeals(fetched)
   953→      }
   954→    } else if (!res.success) {
   955→      // ISSUE 4: surface API errors as a toast instead of silently leaving

   957→      toast.error(res.error || 'Failed to load deals')
   958→      setDeals([])
   959→    }
   960→    if (!signal?.aborted) setLoading(false)
   961→  }, [selectedCategory, debouncedSearch, radiusEnabled, geo.location, readDealAlertRadius])
   962→
   963→  useEffect(() => {
   964→    // HIGH 8: AbortController + cancelled flag pattern. If the user changes
   965→    // the category or search query while a previous request is still in
   966→    // flight, we abort the stale request and ignore its result.
   967→    const controller = new AbortController()
   968→    fetchDeals(controller.signal)
   969→    return () => controller.abort()
   970→  }, [fetchDeals])
   971→

   973→  // the "Near me" button), re-fetch with the radius filter active.
   974→  useEffect(() => {
   975→    if (geoNonce === 0) return
   976→    const controller = new AbortController()
   977→    fetchDeals(controller.signal, { withRadius: true })
   978→    return () => controller.abort()
   979→  }, [geoNonce, fetchDeals])
   980→
   981→  // ISSUE 2: "Near me" button handler. Requests geolocation, then flips on
   982→  // the radius filter and bumps geoNonce to trigger a re-fetch.
   983→  const handleNearMe = useCallback(() => {
   984→    if (geo.loading) return
   985→    if (geo.location) {

   990→      return
   991→    }

   993→    toast.info('Getting your location…')

   995→    // Watch for the location to come in. Because useGeolocation doesn't
   996→    // expose a promise, we poll the cached location in localStorage.

   998→    const checkInterval = setInterval(() => {
   999→      const elapsed = Date.now() - startedAt
  1000→      // Bail after 12s — the geolocation request likely failed/timed out.
  1001→      if (elapsed > 12000) {

  1659→// Pickup progress slider - animated bar showing time remaining
  1660→function PickupProgressSlider({ pickupDeadline, createdAt }: { pickupDeadline: string; createdAt?: string }) {
  1661→  const [progress, setProgress] = useState(0)
  1662→  const [timeLeft, setTimeLeft] = useState('')
  1663→
  1664→  useEffect(() => {
  1665→    const calcProgress = () => {
  1666→      const deadline = new Date(pickupDeadline).getTime()
  1667→      // MEDIUM 11: use the actual order createdAt (passed in) to compute the
  1668→      // pickup window, instead of assuming a fixed 2-hour window. Falls back
  1669→      // to the legacy 2-hour assumption if createdAt is not provided.
  1670→      const totalWindow = createdAt
  1671→        ? Math.max(60_000, deadline - new Date(createdAt).getTime())
  1672→        : 2 * 60 * 60 * 1000
  1673→      const created = deadline - totalWindow
  1674→      const now = Date.now()
  1675→      const elapsed = now - created
  1676→      const pct = Math.min(100, Math.max(0, (elapsed / totalWindow) * 100))
  1677→      setProgress(pct)
  1678→
  1679→      const remaining = deadline - now
  1680→      if (remaining <= 0) {
  1681→        // Pickup window has passed — but the order may still be "active"
  1682→        // (pending_pickup). Show a clear message instead of "Expired" which
  1683→        // confuses users into thinking the deal/order is gone.
  1684→        setTimeLeft('Pickup overdue')
  1685→      } else {
  1686→        const mins = Math.floor(remaining / 60000)
  1687→        const hrs = Math.floor(mins / 60)
  1688→        if (hrs > 0) {
  1689→          setTimeLeft(`${hrs}h ${mins % 60}m left`)
  1690→        } else {
  1691→          setTimeLeft(`${mins}m left`)
  1692→        }
  1693→      }
  1694→    }
  1695→    calcProgress()
  1696→    const interval = setInterval(calcProgress, 30000) // update every 30s
  1697→    return () => clearInterval(interval)
  1698→  }, [pickupDeadline, createdAt])
  1699→
  1700→  const isOverdue = timeLeft === 'Pickup overdue'
  1701→  const isUrgent = !isOverdue && progress > 75
  1702→  const isWarning = !isOverdue && progress > 50
  1703→
  1704→  return (
  1705→    <div className="mt-2">
  1706→      <div className="flex items-center justify-between mb-1">
  1707→        <span className={`text-[10px] font-bold ${isOverdue ? 'text-[#717971]' : isUrgent ? 'text-red-500' : isWarning ? 'text-[#E53935]' : 'text-[#E53935]'}`}>
  1708→          {timeLeft}
  1709→        </span>
  1710→      </div>
  1711→      <div className="h-1.5 bg-[#f0f4f2] rounded-full overflow-hidden">
  1712→        <motion.div
  1713→          initial={{ width: 0 }}
  1714→          animate={{ width: `${progress}%` }}
  1715→          transition={{ duration: 1.2, ease: 'easeOut' }}
  1716→          className={`h-full rounded-full ${
  1717→            isOverdue ? 'bg-[#c1c9c0]' : isUrgent ? 'bg-red-500' : isWarning ? 'bg-[#E53935]' : 'bg-[#E53935]'
  1718→          }`}
  1719→          style={{
  1720→            background: isOverdue
  1721→              ? '#c1c9c0'
  1722→              : isUrgent
  1723→              ? 'linear-gradient(90deg, #E53935, #EF4444)'
  1724→              : isWarning
  1725→              ? 'linear-gradient(90deg, #E53935, #E53935)'
  1726→              : 'linear-gradient(90deg, #EF5350, #E53935)',
  1727→          }}
  1728→        />
  1729→      </div>
  1730→    </div>
  1731→  )
  1732→}
