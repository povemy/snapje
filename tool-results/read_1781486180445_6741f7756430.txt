   232→// ============================================
   233→// COUNTDOWN TIMER COMPONENT
   234→// ============================================
   235→const CountdownTimer = memo(function CountdownTimer({ expiresAt, compact = false }: { expiresAt: string; compact?: boolean }) {
   236→  const [timeLeft, setTimeLeft] = useState('')
   237→
   238→  useEffect(() => {
   239→    const update = () => {
   240→      const diff = new Date(expiresAt).getTime() - Date.now()
   241→      if (diff <= 0) {
   242→        setTimeLeft('Expired')
   243→        return
   244→      }
   245→      const hours = Math.floor(diff / 3600000)
   246→      const mins = Math.floor((diff % 3600000) / 60000)
   247→      const secs = Math.floor((diff % 60000) / 1000)
   248→
   249→      if (hours > 0) {
   250→        setTimeLeft(`${hours}h ${mins}m`)
   251→      } else if (mins > 0) {
   252→        setTimeLeft(`${mins}m ${secs}s`)
   253→      } else {
   254→        setTimeLeft(`${secs}s`)
   255→      }
   256→    }
   257→    update()
   258→    const interval = setInterval(update, 1000)
   259→    return () => clearInterval(interval)
   260→  }, [expiresAt])
   261→
   262→  const isUrgent = timeLeft !== 'Expired' && !timeLeft.includes('h')
   263→
   264→  return (
   265→    <span className={`inline-flex items-center gap-1 ${compact ? 'text-xs' : 'text-sm'} font-bold ${
   266→      timeLeft === 'Expired' ? 'text-[#70787d]' : isUrgent ? 'text-[#FB923C] animate-pulse-urgent' : 'text-[#0d6683]'
   267→    }`}>
   268→      <Clock className={compact ? 'w-3 h-3' : 'w-3.5 h-3.5'} />
   269→      {timeLeft}
   270→    </span>
   271→  )
   272→})

   483→function DealDetailView() {

   493→    if (!viewParams.id) return

   495→    apiFetch<Deal>(`/api/deals/${viewParams.id}`).then((res) => {

   500→  const handleClaim = async () => {

   504→      return
   505→    }
   506→    setClaiming(true)
   507→    try {
   508→      // Step 1: Claim (create reservation)
   509→      const claimRes = await apiFetch<{ reservation: { id: string } }>(`/api/deals/${deal.id}/claim`, {
   510→        method: 'POST',
   511→        body: JSON.stringify({ quantity: 1 }),
   512→      })
   513→      if (!claimRes.success) {
   514→        toast.error(claimRes.error || 'Failed to claim deal')
