   226→// ============================================
   227→// COUNTDOWN TIMER COMPONENT
   228→// ============================================
   229→function CountdownTimer({ expiresAt, compact = false }: { expiresAt: string; compact?: boolean }) {
   230→  const [timeLeft, setTimeLeft] = useState('')
   231→
   232→  useEffect(() => {
   233→    const update = () => {
   234→      const diff = new Date(expiresAt).getTime() - Date.now()
   235→      if (diff <= 0) {
   236→        setTimeLeft('Expired')
   237→        return
   238→      }
   239→      const hours = Math.floor(diff / 3600000)
   240→      const mins = Math.floor((diff % 3600000) / 60000)
   241→      const secs = Math.floor((diff % 60000) / 1000)
   242→
   243→      if (hours > 0) {
   244→        setTimeLeft(`${hours}h ${mins}m`)
   245→      } else if (mins > 0) {
   246→        setTimeLeft(`${mins}m ${secs}s`)
   247→      } else {
   248→        setTimeLeft(`${secs}s`)
   249→      }
   250→    }
   251→    update()
   252→    const interval = setInterval(update, 1000)
   253→    return () => clearInterval(interval)
   254→  }, [expiresAt])
   255→
   256→  const isUrgent = timeLeft !== 'Expired' && !timeLeft.includes('h')
   257→
   258→  return (
   259→    <span className={`inline-flex items-center gap-1 ${compact ? 'text-xs' : 'text-sm'} font-bold ${
   260→      timeLeft === 'Expired' ? 'text-[#70787d]' : isUrgent ? 'text-[#FB923C] animate-pulse-urgent' : 'text-[#0d6683]'
   261→    }`}>
   262→      <Clock className={compact ? 'w-3 h-3' : 'w-3.5 h-3.5'} />
   263→      {timeLeft}
   264→    </span>
   265→  )
   266→}

   459→function DealDetailView() {
   460→  const { viewParams, goBack } = useAppStore()
   461→  const [deal, setDeal] = useState<Deal | null>(null)
   462→  const [loading, setLoading] = useState(true)
   463→  const [claiming, setClaiming] = useState(false)
   464→  const [claimed, setClaimed] = useState(false)
   465→  const [order, setOrder] = useState<Order | null>(null)
   466→  const { user } = useAuthStore()
   467→
   468→  useEffect(() => {
   469→    if (!viewParams.id) return
   470→    setLoading(true)
   471→    apiFetch<Deal>(`/api/deals/${viewParams.id}`).then((res) => {
   472→      if (res.success && res.data) setDeal(res.data)
   473→    }).finally(() => setLoading(false))
   474→  }, [viewParams.id])
   475→
   476→  const handleClaim = async () => {
   477→    if (!deal || !user) return
   478→    setClaiming(true)
   479→    try {
   480→      // Step 1: Claim (create reservation)
   481→      const claimRes = await apiFetch<{ reservation: { id: string } }>(`/api/deals/${deal.id}/claim`, {
   482→        method: 'POST',
   483→        body: JSON.stringify({ quantity: 1 }),
   484→      })
   485→      if (!claimRes.success) {
   486→        toast.error(claimRes.error || 'Failed to claim deal')
   487→        return
   488→      }
   489→
   490→      const reservationId = claimRes.data?.reservation?.id
   491→
   492→      // Step 2: Confirm reservation → Order
   493→      const confirmRes = await apiFetch<Order>(`/api/deals/${deal.id}/confirm`, {
   494→        method: 'POST',
   495→        body: JSON.stringify({ reservationId }),
   496→      })
   497→      if (confirmRes.success && confirmRes.data) {
   498→        setOrder(confirmRes.data)
   499→        setClaimed(true)
   500→        toast.success('Deal claimed! Check your orders for the QR code.')
   501→
   502→        // Refresh deal to show updated stock
   503→        const refreshRes = await apiFetch<Deal>(`/api/deals/${deal.id}`)
   504→        if (refreshRes.success && refreshRes.data) setDeal(refreshRes.data)
   505→      } else {
   506→        toast.error(confirmRes.error || 'Failed to confirm claim')
   507→      }
   508→    } finally {
   509→      setClaiming(false)
   510→    }
   511→  }
   512→
   513→  if (loading) {
   514→    return (
   515→      <div className="pb-28">
   516→        <Skeleton className="aspect-[16/10] rounded-none" />
   517→        <div className="px-5 py-4 space-y-4">
